export default {
  async fetch(request, env, ctx) {
    const url = new URL(request.url);
    
    // Habilitar CORS para que tu app pueda conectarse
    const corsHeaders = {
      "Access-Control-Allow-Origin": "*",
      "Access-Control-Allow-Methods": "GET, POST, OPTIONS",
      "Access-Control-Allow-Headers": "Content-Type",
    };

    if (request.method === "OPTIONS") {
      return new Response(null, { headers: corsHeaders });
    }

    // Ruta de búsqueda del Marketplace
    if (url.pathname === "/api/marketplace-search" && request.method === "POST") {
      try {
        const body = await request.json();
        const query = body.query || "";
        
        // Aquí va la lógica de búsqueda (Open VSX, GitHub, etc.)
        // Por ahora, una prueba básica:
        const response = await fetch(`https://open-vsx.org/api/-/search?query=${encodeURIComponent(query)}&size=20`);
        const data = await response.json();
        
        return new Response(JSON.stringify({ exito: true, resultados: data.extensions }), {
          headers: { ...corsHeaders, "Content-Type": "application/json" },
        });
      } catch (error) {
        return new Response(JSON.stringify({ exito: false, error: error.message }), {
          headers: { ...corsHeaders, "Content-Type": "application/json" },
          status: 500,
        });
      }
    }

    return new Response("Ruta no encontrada", { status: 404, headers: corsHeaders });
  },
};
